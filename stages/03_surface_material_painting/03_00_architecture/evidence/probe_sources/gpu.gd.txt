extends SceneTree
var failures := 0
var rows: Array = []
var view: SubViewport
var mesh: MeshInstance3D
var shader_material: ShaderMaterial
var reference: ShaderMaterial
var mask: Image
var texture: ImageTexture
func _initialize() -> void:
	call_deferred("run")
func check(ok: bool,name: String,details: Variant=null) -> void:
	print(("PASS " if ok else "FAIL ")+name+" "+JSON.stringify(details))
	rows.append({"name":name,"status":"PASS" if ok else "FAIL","details":details})
	if not ok: failures+=1
func solid(color: Color) -> ImageTexture:
	var image:=Image.create(4,4,false,Image.FORMAT_RGBA8)
	image.fill(color)
	return ImageTexture.create_from_image(image)
func pixel(material: Material) -> Color:
	mesh.material_override=material
	for i in 5: await process_frame
	await RenderingServer.frame_post_draw
	return view.get_texture().get_image().get_pixel(128,128)
func compare(name: String,expected: Color) -> void:
	var observed: Color=await pixel(shader_material)
	reference.set_shader_parameter("expected",Vector3(expected.r,expected.g,expected.b))
	var baseline: Color=await pixel(reference)
	var error:=maxf(absf(observed.r-baseline.r),maxf(absf(observed.g-baseline.g),absf(observed.b-baseline.b)))
	check(error<0.035,name,{"observed":str(observed),"reference":str(baseline),"max_error":error})
func run() -> void:
	print("DEVICE ",RenderingServer.get_video_adapter_name()," / ",RenderingServer.get_video_adapter_api_version()," / ",RenderingServer.get_current_rendering_method())
	view=SubViewport.new()
	view.size=Vector2i(256,256)
	view.own_world_3d=true
	view.render_target_update_mode=SubViewport.UPDATE_ALWAYS
	root.add_child(view)
	var environment:=WorldEnvironment.new()
	environment.environment=Environment.new()
	environment.environment.background_mode=Environment.BG_COLOR
	environment.environment.background_color=Color.BLACK
	environment.environment.tonemap_mode=Environment.TONE_MAPPER_LINEAR
	view.add_child(environment)
	var camera:=Camera3D.new()
	camera.projection=Camera3D.PROJECTION_ORTHOGONAL
	camera.size=2.1
	view.add_child(camera)
	camera.position=Vector3(0,3,0)
	camera.look_at(Vector3.ZERO,Vector3.FORWARD)
	camera.current=true
	mesh=MeshInstance3D.new()
	var plane:=PlaneMesh.new()
	plane.size=Vector2(2,2)
	mesh.mesh=plane
	view.add_child(mesh)
	shader_material=ShaderMaterial.new()
	var shader:=Shader.new()
	shader.code=FileAccess.get_file_as_string("res://mix.gdshader")
	shader_material.shader=shader
	reference=ShaderMaterial.new()
	var reference_shader:=Shader.new()
	reference_shader.code="shader_type spatial; render_mode unshaded,cull_disabled; uniform vec3 expected; void fragment(){ALBEDO=expected;}"
	reference.shader=reference_shader
	var colors: Array[Color]=[Color.RED,Color.GREEN,Color.BLUE,Color.WHITE]
	for i in 4:
		shader_material.set_shader_parameter("albedo%d"%i,solid(colors[i]))
		shader_material.set_shader_parameter("normal%d"%i,solid(Color(0.5,0.5,1)))
		shader_material.set_shader_parameter("rough%d"%i,solid(Color(0.2+0.2*i,0,0)))
	mask=Image.create(8,8,false,Image.FORMAT_RGBA8)
	mask.fill(Color(1,0,0,0))
	texture=ImageTexture.create_from_image(mask)
	shader_material.set_shader_parameter("weights",texture)
	for i in 4:
		var weight:=Color(0,0,0,0)
		weight[i]=1.0
		mask.fill(weight)
		texture.update(mask)
		await compare("four_layer_channel_%d"%i,colors[i])
	mask.fill(Color(0,0,0,0))
	texture.update(mask)
	await compare("zero_weights_base_fallback",Color.RED)
	mask.fill(Color8(128,127,0,0))
	texture.update(mask)
	await compare("normalized_weight_blend",Color(128.0/255.0,127.0/255.0,0))
	shader_material.set_shader_parameter("debug_view",1)
	await compare("linear_roughness_blend",Color(0.2*128.0/255.0+0.4*127.0/255.0,0.2*128.0/255.0+0.4*127.0/255.0,0.2*128.0/255.0+0.4*127.0/255.0))
	shader_material.set_shader_parameter("debug_view",2)
	var n:=Vector3(1.0/255.0,1.0/255.0,1).normalized()*0.5+Vector3.ONE*0.5
	await compare("decoded_normal_mix_flat_fixture",Color(n.x,n.y,n.z))
	shader_material.set_shader_parameter("debug_view",0)
	shader_material.set_shader_parameter("detail_tiling",Vector2(23,7))
	await compare("detail_tiling_does_not_change_constant_mask",Color(128.0/255.0,127.0/255.0,0))
	mask.fill(Color(1,0,0,0))
	check(mask.save_png("res://out/gpu_mask.png")==OK,"gpu_fixture_png_saved")
	var disk_before:=FileAccess.get_sha256("res://out/gpu_mask.png")
	mask.fill(Color(0,1,0,0))
	texture.update(mask)
	await compare("ImageTexture_update_changes_GPU",Color.GREEN)
	check(FileAccess.get_sha256("res://out/gpu_mask.png")==disk_before and Image.load_from_file("res://out/gpu_mask.png").get_pixel(0,0).r==1,"GPU_update_does_not_write_source_PNG")
	var data: Resource=load("res://mask.gd").new()
	data.size=Vector2i(8,8)
	data.pixels=mask.get_data()
	var facade: Texture2D=load("res://facade.gd").new()
	facade.data=data
	shader_material.set_shader_parameter("weights",facade)
	await compare("CPU_resource_Texture2D_facade_GPU",Color.GREEN)
	var material_path:="res://out/managed_%s.tres"%RenderingServer.get_current_rendering_method()
	check(ResourceSaver.save(shader_material,material_path)==OK,"facade_managed_material_save")
	var facade_path:="res://out/facade_%s.tres"%RenderingServer.get_current_rendering_method()
	check(ResourceSaver.save(facade,facade_path)==OK,"facade_only_resource_save")
	var saved_text:=FileAccess.get_file_as_string(facade_path)
	check(saved_text.contains("pixels =") and not saved_text.contains('type="ImageTexture"'),"facade_does_not_serialize_GPU_cache")
	var restored:=ResourceLoader.load(material_path,"",ResourceLoader.CACHE_MODE_IGNORE_DEEP) as ShaderMaterial
	check(restored.get_shader_parameter("weights").data.pixels==data.pixels,"facade_saved_authoritative_bytes")
	shader_material=restored
	await compare("facade_reload_GPU_rebuild",Color.GREEN)
	await pixel(shader_material)
	var renderer:=RenderingServer.get_current_rendering_method()
	view.get_texture().get_image().save_png("res://out/gpu_%s.png"%renderer)
	var file:=FileAccess.open("res://out/gpu_%s.json"%renderer,FileAccess.WRITE)
	file.store_string(JSON.stringify({"renderer":renderer,"device":RenderingServer.get_video_adapter_name(),"results":rows,"scope":"13 sampler API and flat unshaded debug values; no lighting/performance/mobile hardware validation","INTERACTIVE_GUI_QA":"NOT_RUN"},"  "))
	file.close()
	view.queue_free()
	await process_frame
	print(("PASS " if failures==0 else "FAIL ")+"GPU_COMPLETE")
	quit(0 if failures==0 else 1)
