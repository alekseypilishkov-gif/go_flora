shader_type spatial;
render_mode unshaded, cull_disabled;
uniform sampler2D weights : filter_linear, repeat_disable;
uniform sampler2D albedo0 : source_color, filter_linear, repeat_enable;
uniform sampler2D albedo1 : source_color, filter_linear, repeat_enable;
uniform sampler2D albedo2 : source_color, filter_linear, repeat_enable;
uniform sampler2D albedo3 : source_color, filter_linear, repeat_enable;
uniform sampler2D normal0 : hint_normal, filter_linear, repeat_enable;
uniform sampler2D normal1 : hint_normal, filter_linear, repeat_enable;
uniform sampler2D normal2 : hint_normal, filter_linear, repeat_enable;
uniform sampler2D normal3 : hint_normal, filter_linear, repeat_enable;
uniform sampler2D rough0 : filter_linear, repeat_enable;
uniform sampler2D rough1 : filter_linear, repeat_enable;
uniform sampler2D rough2 : filter_linear, repeat_enable;
uniform sampler2D rough3 : filter_linear, repeat_enable;
uniform vec2 detail_tiling = vec2(2.0);
uniform int debug_view = 0;
varying vec2 mask_uv;
void vertex() { mask_uv = VERTEX.xz / 2.0 + vec2(0.5); }
void fragment() {
    vec4 w = max(texture(weights,mask_uv),vec4(0.0));
    float sum_w = dot(w,vec4(1.0));
    if (sum_w < 0.00001) { w=vec4(1.0,0.0,0.0,0.0); } else { w/=sum_w; }
    vec2 detail_uv=UV*detail_tiling;
    vec3 c=texture(albedo0,detail_uv).rgb*w.r+texture(albedo1,detail_uv).rgb*w.g+texture(albedo2,detail_uv).rgb*w.b+texture(albedo3,detail_uv).rgb*w.a;
    vec3 n=(texture(normal0,detail_uv).rgb*2.0-1.0)*w.r+(texture(normal1,detail_uv).rgb*2.0-1.0)*w.g+(texture(normal2,detail_uv).rgb*2.0-1.0)*w.b+(texture(normal3,detail_uv).rgb*2.0-1.0)*w.a;
    n=length(n)>0.00001 ? normalize(n) : vec3(0.0,0.0,1.0);
    float r=texture(rough0,detail_uv).r*w.r+texture(rough1,detail_uv).r*w.g+texture(rough2,detail_uv).r*w.b+texture(rough3,detail_uv).r*w.a;
    NORMAL_MAP=n*0.5+0.5;
    ROUGHNESS=clamp(r,0.0,1.0);
    ALBEDO=debug_view==1 ? vec3(r) : (debug_view==2 ? n*0.5+0.5 : c);
}
